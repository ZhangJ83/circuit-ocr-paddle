\begin{frame}{Modality Collapse Problem}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{Collapse Pattern}
        \begin{itemize}
          \item \textbf{S200}: ``1,2,3,4,5...'' (pure counting)
          \item \textbf{S400}: ``+3.3V, +3.3V...'' (voltage repetition)
          \item \textbf{S600}: Partial recovery
          \item Small dataset $\rightarrow$ model ignores image
        \end{itemize}
      \end{block}
    \column{0.5\textwidth}
      \begin{block}{Root Cause}
        \begin{itemize}
          \item 1200 samples insufficient for VLM
          \item Model learns shortcut: output frequent patterns
          \item Training loss $\downarrow$ but quality $\downarrow$
          \item Loss-quality mismatch
        \end{itemize}
      \end{block}
  \end{columns}
  \vspace{0.3cm}
  \begin{center}
    \textcolor{sysured}{\textbf{Solution: Inject synthetic text-image pairs to force visual-text alignment}}
  \end{center}
\end{frame}

\begin{frame}{Phase 2: Synthetic Data Anti-Collapse Strategy}
  \begin{columns}[T]
    \column{0.45\textwidth}
      \begin{block}{Method}
        \begin{enumerate}
          \item Generate 300 synthetic text images (6 templates)
          \item Mix at 20\% ratio with circuit data
          \item Total: 1500 training samples
          \item Two configurations tested:
          \begin{itemize}
            \item exp5: Anti-Overfit (lr=1e-5, do=0.1)
            \item exp6: Baseline (lr=2e-5, do=0.05)
          \end{itemize}
        \end{enumerate}
      \end{block}

\section{Key Findings \& Future Work}
% ══════════════════════════════════════════

\begin{frame}{Key Findings}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{What Worked ($\checkmark$)}
        \begin{enumerate}
          \item \textbf{Synthetic data mixing} — 4.2$\times$ CompF1 gain
          \item \textbf{Anti-overfit config} — stable training, no degradation
          \item \textbf{Frozen projector + LoRA} — prevents catastrophic collapse
          \item \textbf{Multi-metric evaluation} — CompF1 + JointF1 + NED + RepRate
          \item \textbf{7-round annotation pipeline} — >99\% label accuracy
        \end{enumerate}
      \end{block}
    \column{0.5\textwidth}
      \begin{block}{What Didn't Work ($\times$)}
        \begin{enumerate}
          \item \textbf{512px high-res} — no benefit over 384px
          \item \textbf{Unfrozen projector} — higher CompF1 but still collapsed
          \item \textbf{Loss-based checkpoint selection} — loss $\neq$ quality
          \item \textbf{1200 samples} — insufficient for JointF1 $>$ 0.05
          \item \textbf{Single CAD source} — limits diversity score (1.3: 2.5/5)
        \end{enumerate}
      \end{block}
  \end{columns}
  \vspace{0.3cm}
  \begin{center}
    \textbf{Core insight:} Synthetic data teaches visual-text alignment. Real schematic diversity needed for generalization.
  \end{center}
\end{frame}

\begin{frame}{Limitations \& Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{Current Limitations}
        \begin{itemize}
          \item JointF1 (0.011) $\ll$ usable threshold (0.05)
          \item 1200 schematics from single source (KiCad)
          \item Only 20 real-camera photos
          \item No degraded samples (blur, tilt, low-light)
          \item Validation set (3 samples) too small
        \end{itemize}
      \end{block}
    \column{0.5\textwidth}
      \begin{block}{Next Steps}
        \begin{enumerate}
          \item \textbf{Scale data}: 3000+ schematics from multiple EDA tools
          \item \textbf{Real photos}: 200+ photographed circuits
          \item \textbf{Degraded samples}: blur, tilt, low-light variants
          \item \textbf{Two-stage training}: synth pre-train $\rightarrow$ real fine-tune
          \item \textbf{Larger LoRA}: try $r=32$, $r=64$
        \end{enumerate}
      \end{block}
