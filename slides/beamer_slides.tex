\section{Motivation \& Background}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Motivation: Why Circuit OCR?}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{block}{Problem}
        \begin{itemize}
          \item Circuit schematics contain dense technical text:
          \begin{itemize}
            \item Component labels: \texttt{R1, C2, U3, J4}
            \item Values: \texttt{10k$\Omega$, 100nF, 3.3V}
            \item Pin numbers, net names, BOM tables
          \end{itemize}
          \item Manual transcription is error-prone and slow
          \item Existing OCR tools (Tesseract, EasyOCR) fail on schematic fonts
        \end{itemize}
      \end{block}

    \column{0.45\textwidth}
      \begin{block}{Our Approach}
        \begin{enumerate}
          \item Fine-tune \textbf{PaddleOCR-VL-0.9B} — a Vision-Language Model
          \item Use \textbf{LoRA} for parameter-efficient training on consumer GPU (RTX 4060 8GB)
          \item Design \textbf{anti-collapse training strategy} for small datasets
          \item Evaluate with \textbf{multi-metric benchmark}
        \end{enumerate}
      \end{block}
  \end{columns}
\end{frame}

\section{Model Architecture}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Model Architecture: PaddleOCR-VL-0.9B}
  \begin{center}
    \begin{tikzpicture}[node distance=1.8cm, auto,
      block/.style={rectangle, draw, minimum width=2.2cm, minimum height=1cm, rounded corners=3pt, font=\small},
      lora/.style={rectangle, draw=sysured, fill=sysured!5, minimum width=2.2cm, minimum height=0.7cm, rounded corners=3pt, font=\footnotesize, dashed}]

      \node[block,fill=sysublue!15] (img) {Circuit Image};
      \node[block,fill=sysublue!15,right=of img] (vis) {SigLIP Vision\\Encoder};
      \node[block,fill=sysublue!15,right=of vis] (proj) {MLP Projector\\(Frozen)};
      \node[block,fill=sysured!15,right=of proj] (llm) {ERNIE-4.5 LLM\\0.9B Params};
      \node[block,fill=phase2color!15,right=of llm] (out) {Text Output};
      \node[lora,below=of llm] (lora_node) {LoRA r=16, $\alpha$=32\\(5.7M trainable)};

      \draw[->,thick] (img) -- (vis);
      \draw[->,thick] (vis) -- (proj);
      \draw[->,thick] (proj) -- (llm);
      \draw[->,thick] (llm) -- (out);
      \draw[<->,thick,sysured] (llm) -- (lora_node);
    \end{tikzpicture}
  \end{center}

  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{LoRA Configuration}
        \begin{itemize}
          \item $r=16$, $\alpha=32$, dropout
          \item Target: Q/K/V/O proj + MLP layers
          \item \textbf{5.73M} trainable / 0.9B total (0.6\%)
          \item Projector \textbf{frozen} (prevents modality collapse)
        \end{itemize}
      \end{block}
    \column{0.5\textwidth}
      \begin{block}{Training Details}
        \begin{itemize}
          \item Optimizer: AdamW, weight\_decay=0.1
          \item LR: LinearWarmup + CosineAnnealing
          \item Grad Accum = 4, Batch Size = 1
          \item Mixed precision: bfloat16
          \item Hardware: RTX 4060 8GB
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\section{Phase 1: Four-Experiment Design}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Phase 1: Experiment Design}
  \begin{table}
    \centering
    \small
    \begin{tabular}{lcccccc}
      \toprule
      \textbf{Experiment} & \textbf{Resolution} & \textbf{LR} & \textbf{Dropout} & \textbf{Epochs} & \textbf{Special} \\
      \midrule
      \rowcolor{blue!5}  exp1 (Baseline)    & 384px & $2\times10^{-5}$ & 0.05 & 2 & — \\
      \rowcolor{blue!3}  exp2 (HiRes)       & \textbf{512px} & $2\times10^{-5}$ & 0.05 & 2 & — \\
      \rowcolor{red!5}   exp3 (Anti-Overfit)& 384px & $1\times10^{-5}$ & \textbf{0.10} & \textbf{3} & Low LR + High DO \\
      \rowcolor{green!5} exp4 (Unfrozen)    & 384px & $2\times10^{-5}$ & 0.05 & 2 & \textbf{Projector unfrozen} \\
      \bottomrule
    \end{tabular}
  \end{table}

  \begin{block}{Motivation}
    \begin{itemize}
      \item \textbf{exp3}: Hypothesis — lower LR + higher dropout prevents overfitting on 1200 samples
      \item \textbf{exp4}: Hypothesis — unfreezing projector allows better visual-text alignment
      \item \textbf{exp2}: Hypothesis — 512px captures fine text details (pin numbers, small labels)
    \end{itemize}
  \end{block}

  \begin{center}
    \small Data: 1200 clean circuit schematics (1080 train + 120 val)
  \end{center}
\end{frame}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Phase 1: Results — Component F1}
  \begin{center}
    \includegraphics[width=0.82\textwidth]{../slides_figures/comp_f1_comparison.pdf}
  \end{center}

  \begin{block}{Key Finding}
    \begin{itemize}
      \item Best Phase 1 model: \textbf{exp4 (Unfrozen Projector)} — CompF1 = 0.092
      \item All models \textbf{far below} V10-Fixed S600 baseline (CompF1 = 0.206, trained on 3800 samples)
      \item \textbf{Modality Collapse} observed: models output repetitive patterns (``1,2,3,...'') instead of reading images
    \end{itemize}
  \end{block}
\end{frame}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Phase 1: Modality Collapse Analysis}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{Collapse Pattern}
        \begin{itemize}
          \item \textbf{S200}: ``1,2,3,4,5,6,...'' (pure counting)
          \item \textbf{S400}: ``+3.3V, +3.3V, +3.3V...'' (voltage repetition)
          \item \textbf{S600}: ``R1, 10k, 12V...'' (partial recovery)
          \item \textbf{exp3 only}: S800 showed ``R1\textbackslash n10k\textbackslash nR2\textbackslash n10k...''
        \end{itemize}
      \end{block}

    \column{0.5\textwidth}
      \begin{block}{Root Cause}
        \begin{itemize}
          \item Small dataset (1200 samples) — model learns to \textbf{ignore image} and predict frequent patterns
          \item Training loss decreases but validation quality degrades — \textbf{loss-quality mismatch}
          \item Frozen projector may limit visual feature extraction
        \end{itemize}
      \end{block}
  \end{columns}

  \vspace{0.3cm}
  \begin{center}
    \small\textcolor{sysured}{\textbf{→ Solution: Inject synthetic text-image pairs to force visual-text alignment}}
  \end{center}
\end{frame}

\section{Phase 2: Synthetic Data Anti-Collapse}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Synthetic Data Strategy}
  \begin{columns}[T]
    \column{0.45\textwidth}
      \begin{block}{What We Did}
        \begin{enumerate}
          \item Generated \textbf{300 synthetic text images}
          \item Content: component tables, pinouts, net labels, BOM entries
          \item Rendered as clean document-style images (white background, black text)
          \item Mixed with original 1200 circuit images at \textbf{20\% ratio}
          \item Total: \textbf{1500 training samples}
        \end{enumerate}
      \end{block}

    \column{0.55\textwidth}
      \begin{block}{Why It Works}
        \begin{itemize}
          \item Synthetic images have \textbf{perfect visual→text mapping}
          \item Model cannot ``guess'' the text — must read the image
          \item Forces the vision encoder to remain active
          \item Text content matches circuit domain (resistors, capacitors, voltages)
          \item Acts as a \textbf{regularizer} against counting collapse
        \end{itemize}
      \end{block}
  \end{columns}

  \vspace{0.3cm}
  \begin{center}
    \begin{tikzpicture}
      \node[draw,fill=blue!5,rounded corners,minimum width=3cm,minimum height=1cm] (a) {1200 Circuit Images};
      \node[draw,fill=orange!10,rounded corners,minimum width=3cm,minimum height=1cm,right=0.5cm of a] (b) {300 Synthetic Text};
      \node[draw,fill=green!10,rounded corners,minimum width=3cm,minimum height=1cm,right=0.5cm of b] (c) {1500 Mixed Dataset};
      \draw[->,thick] (a) -- (b);

    \end{tikzpicture}
  \end{center}
\end{frame}

\section{Key Findings \& Analysis}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Key Findings}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{What Worked}
        \begin{enumerate}
          \item \textbf{Synthetic text data mixing} — single most effective intervention ($4\times$ CompF1 gain)
          \item \textbf{Anti-overfit config} (exp3/exp5): low LR + high dropout gave \textbf{stable} training
          \item \textbf{LoRA + frozen projector} — prevents catastrophic modality collapse
          \item \textbf{Multi-metric evaluation} — CompF1 + JointF1 + NED + RepRate captures different failure modes
        \end{enumerate}
      \end{block}

    \column{0.5\textwidth}
      \begin{block}{What Didn't Work}
        \begin{enumerate}
          \item \textbf{512px high-res} (exp2) — no benefit, similar collapse
          \item \textbf{Unfrozen projector} (exp4) — higher CompF1 but still collapsed on most samples
          \item \textbf{Loss-based checkpoint selection} — lowest loss $\neq$ best quality
          \item \textbf{1200 samples insufficient} — even with synthetic data, joint\_f1 $\ll$ 0.05 target
        \end{enumerate}
      \end{block}
  \end{columns}

  \vspace{0.3cm}
  \begin{center}
    \small\textcolor{sysublue}{\textbf{Core Insight:} Synthetic data teaches visual-text alignment, but real schematic diversity is needed for generalization.}
  \end{center}
\end{frame}

\section{Limitations \& Future Work}

% ═══════════════════════════════════════════════════════════
\begin{frame}{Limitations \& Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{Current Limitations}
        \begin{itemize}
          \item JointF1 (0.011) far below usable threshold (0.05)
          \item NED ($\sim$0.94) indicates poor per-token accuracy
          \item 1200 training samples insufficient for generalization
          \item Real-camera photos not yet integrated
          \item Validation set (3 samples) too small for reliable monitoring
        \end{itemize}
      \end{block}

    \column{0.5\textwidth}
      \begin{block}{Future Directions}
        \begin{enumerate}
          \item \textbf{Scale up data}: collect 3000+ diverse real schematics
          \item \textbf{Real-camera data}: add photographed circuit boards/prints
          \item \textbf{More synthetic variety}: fonts, sizes, layouts, noise
          \item \textbf{Larger LoRA rank}: try $r=32$ or $r=64$
          \item \textbf{Two-stage training}: synthetic pre-train → real fine-tune
          \item \textbf{Better evaluation}: 50+ sample validation set
        \end{enumerate}
      \end{block}
  \end{columns}
\end{frame}
